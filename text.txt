Улучшим Order.js

export class Order {

    constructor(
        customerName,
        phone,
        items,
        total
    ) {

        this.id = Date.now();

        this.customerName = customerName;

        this.phone = phone;

        this.items = items;

        this.total = total;

        this.status = "NEW";

        this.createdAt =
            new Date().toISOString();
    }
}


---

Улучшим Cart.js

Добавим изменение количества.

increase(productId) {

    const item =
        this.items.find(
            i => i.id === productId
        );

    if(item){

        item.qty++;

        this.save();
    }
}

decrease(productId) {

    const item =
        this.items.find(
            i => i.id === productId
        );

    if(!item) return;

    item.qty--;

    if(item.qty <= 0){

        this.remove(productId);
        return;
    }

    this.save();
}

Улучшим UI.js

static renderCart(cart){

    const container =
        document.querySelector("#cart");

    container.innerHTML = "";

    cart.items.forEach(item=>{

        const div =
            document.createElement("div");

        div.innerHTML = `
            <span>
                ${item.name}
            </span>

            <button
                class="minus"
                data-id="${item.id}"
            >
                -
            </button>

            ${item.qty}

            <button
                class="plus"
                data-id="${item.id}"
            >
                +
            </button>

            =
            ${item.price * item.qty} ₽
        `;

        container.append(div);
    });

    document.querySelector(
        "#total"
    ).textContent =
        cart.getTotal();
}
